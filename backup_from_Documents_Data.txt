#!/bin/bash

# 1. Nhap ten thu muc backup
read -p "Nhap ten thu muc muon luu tai Desktop: " backup_dir
DEST_PATH="$HOME/Desktop/$backup_dir"
SOURCE_PATH="$HOME/Documents/Data"

# Kiem tra xem thu muc goc co ton tai khong
if [ ! -d "$SOURCE_PATH" ]; then
    echo "Loi: Thu muc nguon $SOURCE_PATH khong ton tai!"
    exit 1
fi

# 2. Kiem tra va tao thu muc dich
if [ ! -d "$DEST_PATH" ]; then
    echo "Thu muc $backup_dir chua ton tai. Vui long tien hanh tao thu muc."
    mkdir -p "$DEST_PATH"
fi

# 3. Lay gio hien tai HH:MM:SS
current_time=$(date +%H:%M:%S)
FILE_NAME="Backup $current_time.tar"

# 4. Thuc hien nen va kiem tra ket qua
tar -cf "$DEST_PATH/$FILE_NAME" -C "$HOME/Documents" Data

if [ $? -eq 0 ]; then
    echo "Sao luu du lieu thanh cong!"
else
    echo "Sao luu du lieu THAT BAI!"
fi
